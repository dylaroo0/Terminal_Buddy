### **3. MCP Abstraction Layer** (Critical for Modularity)

```typescript
// hooks/useMCP.ts
import { invoke } from '@tauri-apps/api/core';

export function useMCP() {
  const callTool = async <T>(tool: string, params: Record<string, unknown>): Promise<T> => {
    const response = await invoke('mcp_call_tool', { tool, params });
    return response as T;
  };

  return {
    getHistory: (query: string, limit?: number) => 
      callTool<CommandRecord[]>('buddy_history', { query, limit }),
    
    getSessions: (limit?: number) => 
      callTool<SessionRecord[]>('buddy_sessions', { limit }),
    
    getSystemHealth: () => 
      callTool<SystemSnapshot>('buddy_system_health', {}),
    
    runCommand: (command: string, workingDir?: string) => 
      callTool<CommandRecord>('buddy_run', { command, working_dir: workingDir }),
    
    explain: (query: string, context?: string) => 
      callTool<string>('buddy_explain', { query, context }),
  };
}
```

**Why this matters**: 
- ✅ Swap MCP for direct API calls later without changing components
- ✅ Mock easily for development/testing
- ✅ Same hook works in Tauri app AND MCP App iframe

## 🔄 **State Management Strategy**

### **Use Zustand for Lightweight, Modular State**

```typescript
// store/journalStore.ts
import { create } from 'zustand';

interface JournalState {
  sessions: SessionRecord[];
  commands: Record<string, CommandRecord[]>;
  currentView: 'timeline' | 'topics' | 'search';
  selectedSession: string | null;
  
  // Actions
  loadSessions: () => Promise<void>;
  loadCommands: (sessionId: string) => Promise<void>;
  setView: (view: 'timeline' | 'topics' | 'search') => void;
}

export const useJournalStore = create<JournalState>((set, get) => ({
  sessions: [],
  commands: {},
  currentView: 'timeline',
  selectedSession: null,
  
  loadSessions: async () => {
    const sessions = await useMCP().getSessions();
    set({ sessions });
  },
  
  loadCommands: async (sessionId) => {
    const commands = await useMCP().getHistory('', 100);
    set((state) => ({
      commands: { ...state.commands, [sessionId]: commands }
    }));
  },
  
  setView: (view) => set({ currentView: view }),
}));
```

**Why Zustand over Redux?**
- ✅ Less boilerplate
- ✅ Better TypeScript inference
- ✅ Easy to split into multiple stores (journal, settings, voice, etc.)
- ✅ Works with React Server Components if needed later

### **3. MCP Abstraction Layer** (Critical for Modularity)

```typescript
// hooks/useMCP.ts
import { invoke } from '@tauri-apps/api/core';

export function useMCP() {
  const callTool = async <T>(tool: string, params: Record<string, unknown>): Promise<T> => {
    const response = await invoke('mcp_call_tool', { tool, params });
    return response as T;
  };

  return {
    getHistory: (query: string, limit?: number) => 
      callTool<CommandRecord[]>('buddy_history', { query, limit }),
    
    getSessions: (limit?: number) => 
      callTool<SessionRecord[]>('buddy_sessions', { limit }),
    
    getSystemHealth: () => 
      callTool<SystemSnapshot>('buddy_system_health', {}),
    
    runCommand: (command: string, workingDir?: string) => 
      callTool<CommandRecord>('buddy_run', { command, working_dir: workingDir }),
    
    explain: (query: string, context?: string) => 
      callTool<string>('buddy_explain', { query, context }),
  };
}
```

**Why this matters**: 
- ✅ Swap MCP for direct API calls later without changing components
- ✅ Mock easily for development/testing
- ✅ Same hook works in Tauri app AND MCP App iframe

## 🔄 **State Management Strategy**

### **Use Zustand for Lightweight, Modular State**

```typescript
// store/journalStore.ts
import { create } from 'zustand';

interface JournalState {
  sessions: SessionRecord[];
  commands: Record<string, CommandRecord[]>;
  currentView: 'timeline' | 'topics' | 'search';
  selectedSession: string | null;
  
  // Actions
  loadSessions: () => Promise<void>;
  loadCommands: (sessionId: string) => Promise<void>;
  setView: (view: 'timeline' | 'topics' | 'search') => void;
}

export const useJournalStore = create<JournalState>((set, get) => ({
  sessions: [],
  commands: {},
  currentView: 'timeline',
  selectedSession: null,
  
  loadSessions: async () => {
    const sessions = await useMCP().getSessions();
    set({ sessions });
  },
  
  loadCommands: async (sessionId) => {
    const commands = await useMCP().getHistory('', 100);
    set((state) => ({
      commands: { ...state.commands, [sessionId]: commands }
    }));
  },
  
  setView: (view) => set({ currentView: view }),
}));
```

**Why Zustand over Redux?**
- ✅ Less boilerplate
- ✅ Better TypeScript inference
- ✅ Easy to split into multiple stores (journal, settings, voice, etc.)
- ✅ Works with React Server Components if needed later

class MCPClient {

### Step 1: Create the Template Loader (Copy-Paste)

Create this file: `buddy-app/src/utils/templateLoader.ts`

```typescript
// utils/templateLoader.ts
export type TemplateConfig = {
  name: string;
  design_tokens: {
    font_ui: string;
    font_mono: string;
    font_size_ui: number;
    bg_deep: string;
    bg_surface: string;
    primary: string;
    contrast_mode: 'high' | 'medium' | 'low';
  };
};

const DEFAULT_TEMPLATE: TemplateConfig = {
  name: "Minimal Focus",
  design_tokens: {
    font_ui: "Atkinson Hyperlegible",
    font_mono: "JetBrains Mono",
    font_size_ui: 16,
    bg_deep: "#0F172A",
    bg_surface: "#1E293B",
    primary: "#3B82F6",
    contrast_mode: "high",
  },
};

export function applyTemplate(config: TemplateConfig) {
  const root = document.documentElement;
  root.style.setProperty('--font-ui', config.design_tokens.font_ui);
  root.style.setProperty('--font-mono', config.design_tokens.font_mono);
  root.style.setProperty('--font-size-ui', `${config.design_tokens.font_size_ui}px`);
  root.style.setProperty('--bg-deep', config.design_tokens.bg_deep);
  root.style.setProperty('--bg-surface', config.design_tokens.bg_surface);
  root.style.setProperty('--primary', config.design_tokens.primary);
  
  // High contrast = WCAG AAA
  if (config.design_tokens.contrast_mode === 'high') {
    root.style.setProperty('--text-primary', '#F1F5F9');
    root.style.setProperty('--text-secondary', '#94A3B8');
  }
}

export function resetToAccessibleDefault() {
  applyTemplate(DEFAULT_TEMPLATE);
}
```
